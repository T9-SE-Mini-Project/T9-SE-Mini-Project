48def seed_demo(db: Session) -> list[tuple[str, str]]: 

49 """Create the demo data. Returns (email, role) for every demo account.""" 

50 accounts = [(email, role) for email, role, _ in DEMO_ACCOUNTS] 

51 if AccountRepository(db).get_by_email(DEMO_ACCOUNTS[0][0]) is not None: 

52 return accounts

54 created = {} 

55 for email, role, hospital_id in DEMO_ACCOUNTS:

62 is_verified=True, 

63 ) 

64 db.add(account)

66 created[role] = account 

67 db.commit()

69 today = date.today() 

70 donor = created["donor"]
