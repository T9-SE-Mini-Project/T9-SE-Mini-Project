1"""Blood groups, components and transfusion compatibility shared by every module (BLOOD-16). 

2 

3This used to live in hospital_requests, with copies of the constants in donor and 

4inventory. Keeping one copy means the modules cannot disagree about what a valid 

5blood group is, or about which units a patient may receive. 

6 

7Pure functions, no I/O. `compatible_donor_groups(recipient, component)` answers: which 

8donor blood groups may be given to a patient of this group, for this component? 

9 

10Rules used (standard transfusion practice, kept deliberately simple): 

11 - RBC: standard red-cell table (O- is the universal donor, AB+ the universal recipient). 

12 - Whole Blood: identical ABO and Rh only. 

13 - Plasma: reverse of RBC for ABO (AB plasma is universal), Rh is ignored. 

14 - Platelets: identical ABO; an Rh-negative patient only gets Rh-negative, an Rh-positive 

15 patient may get either. 

16""" 

17from __future__ import annotations 

18 

19from typing import Literal, get_args 

20 

21BloodGroup = Literal["A+", "A-", "B+", "B-", "AB+", "AB-", "O+", "O-"] 

22Component = Literal["Whole Blood", "RBC", "Platelets", "Plasma"] 

23 

24BLOOD_GROUPS: tuple[str, ...] = get_args(BloodGroup) 

25COMPONENTS: tuple[str, ...] = get_args(Component) 

26 

27_RBC_DONORS: dict[str, frozenset[str]] = { 

28 "O-": frozenset({"O-"}), 

29 "O+": frozenset({"O+", "O-"}), 

30 "A-": frozenset({"A-", "O-"}), 

31 "A+": frozenset({"A+", "A-", "O+", "O-"}), 

32 "B-": frozenset({"B-", "O-"}), 

33 "B+": frozenset({"B+", "B-", "O+", "O-"}), 

34 "AB-": frozenset({"AB-", "A-", "B-", "O-"}), 

35 "AB+": frozenset(BLOOD_GROUPS), 

36} 

37 

38# Recipient ABO -> ABO types whose plasma they can receive. 

39_PLASMA_DONOR_ABO: dict[str, tuple[str, ...]] = { 

40 "O": ("O", "A", "B", "AB"), 

41 "A": ("A", "AB"), 

42 "B": ("B", "AB"), 

43 "AB": ("AB",), 

44} 

45 

46 

47def _split(group: str) -> tuple[str, str]: 

48 if group not in BLOOD_GROUPS: 

49 raise ValueError(f"Unknown blood group: {group!r}") 

50 return group[:-1], group[-1] 

51 

52 

53def compatible_donor_groups(recipient: str, component: str) -> frozenset[str]: 

54 """Donor blood groups that may be given to `recipient` for `component`.""" 

55 abo, rh = _split(recipient) 

56 if component == "Whole Blood": 

57 return frozenset({recipient}) 

58 if component == "RBC": 

59 return _RBC_DONORS[recipient] 

60 if component == "Plasma": 

61 return frozenset(f"{d}{r}" for d in _PLASMA_DONOR_ABO[abo] for r in "+-") 

62 if component == "Platelets": 

63 rhs = "-" if rh == "-" else "+-" 

64 return frozenset(f"{abo}{r}" for r in rhs) 

65 raise ValueError(f"Unknown component: {component!r} (expected one of {', '.join(COMPONENTS)})") 

66 

67 

68def is_compatible(donor: str, recipient: str, component: str) -> bool: 

69 """True if a `donor` group unit of `component` may be given to a `recipient`.""" 

70 return donor in compatible_donor_groups(recipient, component)
